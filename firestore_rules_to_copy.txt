Firebase Firestore Security Rules

To enforce the daily limit of 5 uploads per user and secure your database, please open your Firebase Console, navigate to Firestore Database -> Rules, and merge these rules into your existing rules:

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Existing rules...
    
    // --- NEW RULES FOR FILE UPLOADS ---
    
    match /submissions/{submissionId} {
      // Only you (the admin) can read all submissions. 
      // Users can only create them.
      allow read: if request.auth != null && request.auth.token.email == "YOUR_ADMIN_EMAIL@studentnotes.com"; 
      allow create: if request.auth != null;
    }
    
    match /users/{userId}/limits/today {
      // Users can only read and write their own limit counters
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // ----------------------------------
  }
}
```
